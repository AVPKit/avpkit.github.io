242  int32_t
243  ContainerFormat :: getOutputNumCodecsSupported()
244  {
245  if (!mOutputFormat)
246  return 0;
247 
248  const struct AVCodecTag * const*tags = mOutputFormat->codec_tag;
249  if (!tags)
250  return 0;
251 
252  int numCodecs = 0;
253 
254  for(int i = 0;
255  tags[i];
256  i++)
257  {
258  for(const struct AVCodecTag * tag = tags[i];
259  tag && tag->id != ICodec::AV_CODEC_ID_NONE;
260  ++tag)
261  {
262  ++numCodecs;
263  }
264  }
265  return numCodecs;
266  }
