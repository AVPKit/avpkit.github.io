101  int FilterChain::configure() {
102  int retval = avfilter_graph_config(mFilterGraph, NULL);
103  if (retval >= 0){
104  for (auto it : mFilters){
105  MediaFilter* filter = dynamic_cast<MediaFilter*>(it.value());
106  filter->setReady();
107  }
108  char* dump = avfilter_graph_dump(mFilterGraph, NULL);
109  if (dump) {
110  VS_LOG_WARN("Filter Graph configured %s", dump);
111  av_free(dump);
112  }
113  }
114 
115  return retval;
116  }
