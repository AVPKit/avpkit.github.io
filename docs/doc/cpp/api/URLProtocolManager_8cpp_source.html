36 const char *
37 URLProtocolManager::parseProtocol(
38  char * dest,
39  int32_t dest_len,
40  const char* url)
41 {
42  if (!dest || dest_len <=0 || !url || !*url)
43  return 0;
44  size_t url_len = strlen(url);
45  if (url_len <= 1)
46  return 0;
47 
48  size_t proto_len = strspn(url, URLProtocolManager_VALID_PROTOCOL_CHARS);
49  if (proto_len <= 0)
50  return 0;
51 
52  // file protocol
53  if (url[proto_len] != ':' && url[proto_len] != ',') {
54  return 0;
55  }
56  if (url[0] && url[1] == ':') {
57  return 0;
58  }
59  if (proto_len >= (size_t)dest_len)
60  // not enough space
61  return 0;
62  strncpy(dest, url, proto_len);
63  dest[proto_len] = 0;
64 
65  return dest;
66 }

118 URLProtocolManager :: URLProtocolManager(
119  const char * aProtocolName)
120 {
121  if (aProtocolName == NULL || *aProtocolName == 0)
122  throw invalid_argument("bad argument to protocol manager");
123  size_t len = strlen(aProtocolName);
124  size_t validChars = strspn(aProtocolName, URLProtocolManager_VALID_PROTOCOL_CHARS);
125  if (validChars != len)
126  throw invalid_argument("protocol name contains invalid characters");
127 
128  mProtoName = new char[len+1];
129  strcpy(mProtoName, aProtocolName);
130  mNext = 0;

138 URLProtocolManager*
139 URLProtocolManager :: findProtocol(const char* protocol,
140  const char* url,
141  int,
142  URLProtocolManager *previous)
143 {
144  char proto[256];
145  if (!protocol || !*protocol) {
146  // parse the protocol from the URL
147  protocol = URLProtocolManager::parseProtocol(proto, sizeof(proto), url);
148  if (!protocol || !*protocol)
149  return 0;
150  }
151  URLProtocolManager *p = previous ? previous : mFirstProtocol;
152  while(p && strcmp(protocol, p->mProtoName) != 0)
153  p = p->mNext;
154  return p;
155 }
