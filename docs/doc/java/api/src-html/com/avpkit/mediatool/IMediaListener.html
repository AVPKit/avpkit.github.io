024import com.avpkit.mediatool.event.AddStreamEvent;

026import com.avpkit.mediatool.event.CloseEvent;
027import com.avpkit.mediatool.event.FlushEvent;
028import com.avpkit.mediatool.event.IAddStreamEvent;
029import com.avpkit.mediatool.event.IAudioSamplesEvent;

050/**
051 * Called by {@link IMediaGenerator} in response to defined events.
052 * 
053 * <p>
054 * 
055 * You can use objects that implement this interface to listen to events
056 * generated by an {@link IMediaGenerator} object. To do that, create your
057 * {@link IMediaListener} object and attach it to an {@link IMediaGenerator} by
058 * calling {@link IMediaGenerator#addListener(IMediaListener)}.
059 * 
060 * </p>
061 * <p>
062 * 
063 * These methods block the calling {@link IMediaGenerator} while they process so
064 * try to return quickly. If you have long running actions to perform, use a
065 * separate thread.
066 * 
067 * </p>
068 */
069
070public interface IMediaListener
071{
072
073  /**
074   * Called after a video picture has been decoded by a {@link IMediaReader} or
075   * encoded by a {@link IMediaWriter}.
076   * 
077   * @param event An event containing either an {@link IVideoPicture}, a
078   *        {@link BufferedImage}, or both. The {@link IVideoPicture} is only
079   *        valid for the duration of this callback. If you need to access it
080   *        after you return, you must either copy the data into your own
081   *        buffers, or call {@link IVideoPicture#copyReference()} to get your
082   *        own reference to the data.
083   */
084
085  public void onVideoPicture(IVideoPictureEvent event);
086
087  /**
088   * Called after audio samples have been decoded or encoded by an
089   * {@link IMediaGenerator}
090   * 
091   * @param event An event containing the {@link IAudioSamples} for this event.
092   *        The {@link IAudioSamples} in this event is only valid for the
093   *        duration of this callback. If you need to access it after you
094   *        return, you must either copy the data into your own buffers, or call
095   *        {@link IAudioSamples#copyReference()} to get your own reference to
096   *        the data.
097   */
098
099  public void onAudioSamples(IAudioSamplesEvent event);
100
101  /**
102   * Called after an {@link IMediaGenerator} is opened.
103   * 
104   * @param event A {@link OpenEvent}
105   */
106
107  public void onOpen(IOpenEvent event);
108
109  /**
110   * Called after an {@link IMediaGenerator} is closed.
111   * 
112   * @param event A {@link CloseEvent}
113   */
114
115  public void onClose(ICloseEvent event);
116
117  /**
118   * Called after an stream is added to an {@link IMediaGenerator}. This occurs
119   * when a new stream is added (if writing) or encountered by the
120   * {@link IMediaCoder} (if reading). If the stream is not already been opened,
121   * then {@link #onOpenCoder(IOpenCoderEvent)} will be called at some later
122   * point.
123   * 
124   * @param event A {@link AddStreamEvent} event
125   */
126
127  public void onAddStream(IAddStreamEvent event);
128
129  /**
130   * Called after a decoder or encoder is opened by a {@link IMediaCoder}
131   * 
132   * @param event A {@link OpenCoderEvent}
133   */
134
135  public void onOpenCoder(IOpenCoderEvent event);
136
137  /**
138   * Called after an decoder or encoder is closed by the {@link IMediaCoder}
139   * .
140   * 
141   * @param event A {@link CloseCoderEvent}
142   */
143
144  public void onCloseCoder(ICloseCoderEvent event);
145
146  /**
147   * Called after a {@link IPacket} has been read by a
148   * {@link IMediaReader}.
149   * 
150   * @param event A {@link ReadPacketEvent}. This {@link IPacket} in this event
151   *        is only valid for the duration of this call. If you need to use the
152   *        data after this call has returned, you must either copy the data in
153   *        this call, or use {@link IPacket#copyReference()} to create a new
154   *        object with a reference you can own.
155   */
156
157  public void onReadPacket(IReadPacketEvent event);
158
159  /**
160   * Called after a {@link IPacket} has been written by a
161   * {@link IMediaWriter}.
162   * 
163   * @param event A {@link WritePacketEvent}. The {@link IPacket} in this event
164   *        is only valid for the duration of this call. If you need to use the
165   *        data after this call has returned, you must either copy the data in
166   *        this call, or use {@link IPacket#copyReference()} to create a new
167   *        object with a reference you can own.
168   */
169
170  public void onWritePacket(IWritePacketEvent event);
171
172  /**
173   * Called after a {@link IMediaWriter} writes the header.
174   * 
175   * @param event A {@link WriteHeaderEvent}
176   */
177
178  public void onWriteHeader(IWriteHeaderEvent event);
179
180  /**
181   * Called after a {@link IMediaWriter} has flushed its buffers.
182   * 
183   * @param event A {@link FlushEvent}
184   */
185
186  public void onFlush(IFlushEvent event);
187
188  /**
189   * Called after a {@link IMediaWriter} writes the trailer.
190   * 
191   * @param event A {@link WriteTrailerEvent}
192   */
193
194  public void onWriteTrailer(IWriteTrailerEvent event);
195}
