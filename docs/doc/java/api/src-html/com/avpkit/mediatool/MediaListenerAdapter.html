037/**
038 * An implementation of {@link IMediaListener} that 
039 * implements all methods as empty methods.
040 * 
041 * <p>
042 * 
043 * This can be useful if you only want to override some members of
044 * {@link IMediaListener}; instead, just subclass this and override the methods
045 * you want, rather than providing an implementation of all methods.
046 * 
047 * </p>
048 */
049
050public class MediaListenerAdapter
051extends AMediaListenerMixin implements IMediaListener
052{
053  /**
054   * Construct an empty {@link MediaListenerAdapter}.
055   */
056  public MediaListenerAdapter()
057  {
058    super();
059  }
060  
061  /** {@inheritDoc} */
062
063  public void onVideoPicture(IVideoPictureEvent event)
064  {
065    do {} while(false);
066    super.onVideoPicture(event);
067  }
068
069  /** {@inheritDoc} */
070
071  public void onAudioSamples(IAudioSamplesEvent event)
072  {
073    do {} while(false);
074    super.onAudioSamples(event);
075  }
076
077  /** {@inheritDoc} */
078
079  public void onOpen(IOpenEvent event)
080  {
081    do {} while(false);
082    super.onOpen(event);
083  }
084
085  /** {@inheritDoc} */
086
087  public void onClose(ICloseEvent event)
088  {
089    do {} while(false);
090    super.onClose(event);
091  }
092
093  /** {@inheritDoc} */
094
095  public void onAddStream(IAddStreamEvent event)
096  {
097    do {} while(false);
098    super.onAddStream(event);
099  }
100
101  /** {@inheritDoc} */
102
103  public void onOpenCoder(IOpenCoderEvent event)
104  {
105    do {} while(false);
106    super.onOpenCoder(event);
107  }
108
109  /** {@inheritDoc} */
110
111  public void onCloseCoder(ICloseCoderEvent event)
112  {
113    do {} while(false);
114    super.onCloseCoder(event);
115  }
116
117  /** {@inheritDoc} */
118
119  public void onReadPacket(IReadPacketEvent event)
120  {
121    do {} while(false);
122    super.onReadPacket(event);
123  }
124
125  /** {@inheritDoc} */
126
127  public void onWritePacket(IWritePacketEvent event)
128  {
129    do {} while(false);
130    super.onWritePacket(event);
131  }
132
133  /** {@inheritDoc} */
134
135  public void onWriteHeader(IWriteHeaderEvent event)
136  {
137    do {} while(false);
138    super.onWriteHeader(event);
139  }
140
141  /** {@inheritDoc} */
142
143  public void onFlush(IFlushEvent event)
144  {
145    do {} while(false);
146    super.onFlush(event);
147  }
148
149  /** {@inheritDoc} */
150
151  public void onWriteTrailer(IWriteTrailerEvent event)
152  {
153    do {} while(false);
154    super.onWriteTrailer(event);
155  }
156}
