020package com.avpkit.mediatool.demos;
021
022
023import com.avpkit.mediatool.IMediaReader;
024import com.avpkit.mediatool.IMediaViewer;
025import com.avpkit.mediatool.ToolFactory;
026
027/**
028 * Using {@link IMediaReader}, takes a media container, finds the first audio
029 * stream, decodes that stream, and plays the audio on your speakers.
030 * 
031 * @author aclarke
032 * @author trebor
033 */
034
035public class DecodeAndPlayAudio
036{
037
038  /**
039   * Takes a media container (file) as the first argument, opens it, opens up a
040   * Swing window and displays video frames with the right
041   * timing.
042   * 
043   * @param args
044   *          Must contain one string which represents a filename
045   */
046
047  public static void main(String[] args)
048  {
049    if (args.length <= 0)
050      throw new IllegalArgumentException(
051          "must pass in a filename as the first argument");
052
053    // create a media reader for processing video
054
055    IMediaReader reader = ToolFactory.makeReader(args[0]);
056
057    // Create a MediaViewer object and tell it to play audio only
058
059    reader.addListener(ToolFactory.makeViewer(IMediaViewer.Mode.AUDIO_ONLY));
060
061    // read out the contents of the media file, and sit back and watch
062
063    while (reader.readPacket() == null)
064      do {} while(false);
065  }
066}
