020package com.avpkit.mediatool.event;
021
022import com.avpkit.mediatool.IMediaCoder;
023import com.avpkit.core.IPacket;
024
025/**
026 * An abstract implementation of {@link IPacketEvent}, but does
027 * not declare {@link IPacketEvent}.
028 * 
029 * @author aclarke
030 *
031 */
032public abstract class APacketMixin extends AStreamCoderMixin
033{
034  private final IPacket mPacket;
035  
036  /**
037   * Create an {@link APacketMixin}.
038   * @param source the source
039   * @param packet a packet.  This event will <strong>not</strong>
040   *   call {@link IPacket#copyReference()} on this packet.
041   */
042  public APacketMixin(IMediaCoder source, 
043      IPacket packet)
044  {
045    super(source, packet == null ? null : packet.getStreamIndex());
046    mPacket = packet;
047  }
048  /**
049   * Implementation of {@link IPacketEvent#getPacket()}.
050   * @see com.avpkit.mediatool.event.IPacketEvent#getPacket()
051   */
052  public IPacket getPacket()
053  {
054    return mPacket;
055  }
056  
057}
